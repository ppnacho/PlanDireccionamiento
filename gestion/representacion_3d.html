/** * NetControl - Motor WebGL 3D para Sala de Servidores (Three.js) * Acceso estrictamente mediante la Edge Function 'obtener-dispositivos'. */ import { supabase } from './supabaseClient.js'; let allDevices = []; let scene, camera, renderer, controls; let rackMeshes = []; // Almacena los objetos 3D interactivos de los racks/blades let raycaster, mouse; document.addEventListener('DOMContentLoaded', () => { verifySession(); loadUserInfo(); init3DEnvironment(); fetchDevices(); }); function verifySession() { const isAuth = localStorage.getItem('netcontrol_auth'); if (!isAuth || !isAuth === 'true') { window.location.href = '../index.html'; } } function loadUserInfo() { const userEmail = localStorage.getItem('netcontrol_user'); if (userEmail) { document.querySelectorAll('.user-email-text').forEach(el => { el.textContent = userEmail; }); } } window.handleLogout = function() { localStorage.removeItem('netcontrol_auth'); localStorage.removeItem('netcontrol_user'); window.location.href = '../index.html'; }; /** * Inicializa el motor gráfico Three.js y la escena de la sala técnica */ function init3DEnvironment() { const container = document.getElementById('webgl-container'); if (!container) return; // 1. Escena y Cámara scene = new THREE.Scene(); scene.background = new THREE.Color(0x020408); scene.fog = new THREE.FogExp2(0x020408, 0.035); camera = new THREE.PerspectiveCamera(60, container.clientWidth / container.clientHeight, 0.1, 1000); camera.position.set(0, 8, 16); // 2. Renderizador WebGL renderer = new THREE.WebGLRenderer({ antialias: true }); renderer.setSize(container.clientWidth, container.clientHeight); renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2)); renderer.shadowMap.enabled = true; container.appendChild(renderer.domElement); // 3. Controles de Órbita controls = new THREE.OrbitControls(camera, renderer.domElement); controls.enableDamping = true; controls.dampingFactor = 0.05; controls.maxPolarAngle = Math.PI / 2 - 0.05; // No pasar por debajo del suelo controls.minDistance = 5; controls.maxDistance = 35; // 4. Iluminación Profesional de Data Center const ambientLight = new THREE.AmbientLight(0x1e293b, 1.5); scene.add(ambientLight); const dirLight = new THREE.DirectionalLight(0x38bdf8, 1.2); dirLight.position.set(10, 20, 10); scene.add(dirLight); const pointLight = new THREE.PointLight(0x3b82f6, 2, 30); pointLight.position.set(0, 10, 0); scene.add(pointLight); // 5. Suelo técnico flotante con rejilla const floorGeo = new THREE.PlaneGeometry(50, 50); const floorMat = new THREE.MeshStandardMaterial({ color: 0x090d16, roughness: 0.8, metalness: 0.2 }); const floor = new THREE.Mesh(floorGeo, floorMat); floor.rotation.x = -Math.PI / 2; floor.position.y = -2; scene.add(floor); const gridHelper = new THREE.GridHelper(50, 35, 0x1d4ed8, 0x1e293b); gridHelper.position.y = -1.99; scene.add(gridHelper); // 6. Raycaster para detección de clics en 3D raycaster = new THREE.Raycaster(); mouse = new THREE.Vector2(); renderer.domElement.addEventListener('click', onCanvasClick); // Evento de redimensionado de ventana window.addEventListener('resize', onWindowResize); // Loop de animación animate(); } function animate() { requestAnimationFrame(animate); controls.update(); renderer.render(scene, camera); } function onWindowResize() { const container = document.getElementById('webgl-container'); if (!container) return; camera.aspect = container.clientWidth / container.clientHeight; camera.updateProjectionMatrix(); renderer.setSize(container.clientWidth, container.clientHeight); } /** * Obtiene los dispositivos mediante la Edge Function */ async function fetchDevices() { try { const { data, error } = await supabase.functions.invoke('obtener-dispositivos', { method: 'GET' }); if (error) throw new Error(error.message || 'Error de comunicación con el servidor.'); if (data && data.success === false) throw new Error(data.message || 'No se pudieron recuperar los dispositivos.'); allDevices = data.devices || []; build3DRacks(allDevices); } catch (err) { console.error('Error al cargar dispositivos 3D:', err); } } /** * Construye dinámicamente las 5 filas de racks en la escena 3D WebGL */ function build3DRacks(devices) { // Limpiar racks anteriores si existieran rackMeshes.forEach(m => scene.remove(m)); rackMeshes = []; const targetPlants = ['BAJA', 'ESCALERA 1', 'PRIMERA', 'ESCALERA 2', 'BUHARDILLA']; const groupedByPlant = devices.reduce((acc, d) => { let planta = (d.planta || 'BAJA').toUpperCase(); if (!acc[planta]) acc[planta] = []; acc[planta].push(d); return acc; }, {}); // Posicionar cada una de las 5 filas en profundidad (eje Z) targetPlants.forEach((plantaName, index) => { const items = groupedByPlant[plantaName] || []; const zPos = (index - 2) * 3.5; // Espaciado en profundidad para las 5 filas // Chasis exterior del Rack 3D const rackGeo = new THREE.BoxGeometry(2.4, 3.2, 0.8); const rackMat = new THREE.MeshStandardMaterial({ color: 0x0f172a, roughness: 0.4, metalness: 0.8 }); const rackMesh = new THREE.Mesh(rackGeo, rackMat); rackMesh.position.set(0, 0, zPos); scene.add(rackMesh); // Si hay dispositivos, colocamos las unidades (Blades) dentro del rack 3D items.forEach((device, dIndex) => { const yOffset = 1.1 - (dIndex * 0.45); // Apilados verticalmente dentro del rack const bladeGeo = new THREE.BoxGeometry(2.2, 0.38, 0.7); const isWifi = device.tipo_conexion === 'WIFI'; const bladeMat = new THREE.MeshStandardMaterial({ color: isWifi ? 0x1e3a8a : 0x065f46, roughness: 0.3, metalness: 0.5 }); const bladeMesh = new THREE.Mesh(bladeGeo, bladeMat); bladeMesh.position.set(0, yOffset, zPos + 0.05); // Asociar datos del dispositivo al objeto 3D para el evento de clic bladeMesh.userData = device; scene.add(bladeMesh); rackMeshes.push(bladeMesh); }); }); } /** * Maneja el clic del ratón sobre los objetos 3D de la escena */ function onCanvasClick(event) { const container = document.getElementById('webgl-container'); if (!container) return; const rect = renderer.domElement.getBoundingClientRect(); mouse.x = ((event.clientX - rect.left) / container.clientWidth) * 2 - 1; mouse.y = -((event.clientY - rect.top) / container.clientHeight) * 2 + 1; raycaster.setFromCamera(mouse, camera); const intersects = raycaster.intersectObjects(rackMeshes); if (intersects.length > 0) { const clickedBlade = intersects[0].object; if (clickedBlade.userData && clickedBlade.userData.id_hw) { openBladeModal(clickedBlade.userData); } } } // Modal interactivo de diagnóstico (Burbuja Blade) window.openBladeModal = function(device) { document.getElementById('modal-title').textContent = device.id_hw || 'Sin ID'; document.getElementById('modal-hostname').textContent = device.hostname ? `@${device.hostname}` : 'Sin hostname'; document.getElementById('modal-ip').textContent = device.ip || 'DHCP Dinámica'; document.getElementById('modal-mac').textContent = device.mac || 'N/D'; document.getElementById('modal-ubicacion').textContent = `${device.planta} / ${device.estancia || 'General'}`; document.getElementById('modal-conexion').textContent = `${device.tipo_conexion} ${device.ssid ? '(' + device.ssid + ')' : ''} [${device.tipo_ip}]`; const extraSection = document.getElementById('modal-extra-section'); let extraHtml = ''; const portsInfo = [ device.puerto_rout_ext ? `Rout Ext: ${device.puerto_rout_ext}➔${device.puerto_rout_int || '?'}` : '', device.puerto_dock_int ? `Dock Int: ${device.puerto_dock_int}` : '' ].filter(Boolean).join(' | '); if (portsInfo || device.dock_network) { extraHtml += `
RED Y PUERTOS ${portsInfo || 'Sin mapeo'} ${device.dock_network ? `
Docker: ${device.dock_network}` : ''}
`; } if ((device.servicio && device.servicio.length > 0) || (device.software && device.software.length > 0)) { extraHtml += `
SERVICIOS & SOFTWARE
${device.servicio ? device.servicio.map(s => `${s}`).join('') : ''} ${device.software ? device.software.map(sw => `${sw}`).join('') : ''}
`; } if (device.comentarios) { extraHtml += `
COMENTARIOS ${device.comentarios}
`; } extraSection.innerHTML = extraHtml; document.getElementById('blade-modal').classList.remove('hidden'); }; window.closeModal = function() { document.getElementById('blade-modal').classList.add('hidden'); };
